cmake_minimum_required(VERSION 3.16)
project(EdgePlateUFPR VERSION 3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(edge_common STATIC shared/src/app.cpp)
target_include_directories(edge_common PUBLIC shared/include)
target_link_libraries(edge_common PUBLIC Threads::Threads)
if(WIN32)
 target_link_libraries(edge_common PUBLIC psapi)
endif()
add_executable(edge_seq baseline/src/main.cpp)
add_executable(edge_pipeline pipeline/src/main.cpp)
target_link_libraries(edge_seq PRIVATE edge_common)
target_link_libraries(edge_pipeline PRIVATE edge_common)
add_executable(kernel_tests tests/kernels.cpp)
target_include_directories(kernel_tests PRIVATE shared/include)
enable_testing()
add_test(NAME known_kernels COMMAND kernel_tests)
